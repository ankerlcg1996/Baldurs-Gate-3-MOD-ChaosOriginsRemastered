PROC
PROC_COS_ConfigEnsureMechanics((CHARACTER)_Character)
AND
DB_COS_ConfigMechanicDefault(_Key, _Default)
AND
NOT DB_COS_ConfigMechanic(_Character, _Key, _)
THEN
DB_COS_ConfigMechanic(_Character, _Key, _Default);

PROC
PROC_COS_ConfigEnableEvents((CHARACTER)_Character)
AND
DB_COS_ConfigMechanicEvent(_Event, _Key)
THEN
EnableTutorialEvent(_Character, _Event);

PROC
PROC_COS_ConfigEnableEvents((CHARACTER)_Character)
AND
DB_COS_ConfigUiOpenedEvent(_Event)
THEN
EnableTutorialEvent(_Character, _Event);

PROC
PROC_COS_ConfigEnableEvents((CHARACTER)_Character)
AND
DB_COS_ConfigResetCoreEvent(_Event)
THEN
EnableTutorialEvent(_Character, _Event);

PROC
PROC_COS_ConfigAddEnabledMechanicMirrors((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic(_Character, _Key, 1)
AND
DB_COS_ConfigMechanicMirror(_Key, _Passive)
AND
HasPassive(_Character, _Passive, 0)
THEN
AddPassive(_Character, _Passive);

PROC
PROC_COS_ConfigRemoveDisabledMechanicMirrors((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic(_Character, _Key, 0)
AND
DB_COS_ConfigMechanicMirror(_Key, _Passive)
AND
HasPassive(_Character, _Passive, 1)
THEN
RemovePassive(_Character, _Passive);

PROC
PROC_COS_ConfigSyncMechanicMirrors((CHARACTER)_Character)
THEN
PROC_COS_ConfigAddEnabledMechanicMirrors(_Character);
PROC_COS_ConfigRemoveDisabledMechanicMirrors(_Character);

PROC
PROC_COS_ConfigSyncCharacter((CHARACTER)_Character)
THEN
PROC_COS_ConfigEnsureMechanics(_Character);
PROC_COS_ConfigEnableEvents(_Character);
PROC_COS_AccrueMastery(_Character);
PROC_COS_SyncMastery(_Character);
PROC_COS_ConfigSyncMechanicMirrors(_Character);

PROC
PROC_COS_ConfigSuspendMastery((CHARACTER)_Character)
THEN
RemoveStatus(_Character, "COS_CHAOS_MASTERY_TUNE", _Character);
RemoveStatus(_Character, "COS_CHAOS_MASTERY_CORRECT", _Character);
RemoveSpell(_Character, "Shout_COS_ChaosMastery", 1);

PROC
PROC_COS_ConfigApplyMechanic((CHARACTER)_Character, (STRING)_Key, (INTEGER)_Enabled)
AND
_Key == "Fate"
AND
_Enabled == 0
THEN
PROC_COS_ClearFateAction(_Character);

PROC
PROC_COS_ConfigApplyMechanic((CHARACTER)_Character, (STRING)_Key, (INTEGER)_Enabled)
AND
_Key == "Power"
AND
_Enabled == 0
THEN
RemoveStatus(_Character, "COS_CHAOS_POWER_STACK", _Character);
RemoveStatus(_Character, "COS_CHAOS_GENESIS_READY", _Character);

THEN

PROC
PROC_COS_ConfigToggleMechanic((CHARACTER)_Character, (STRING)_Key)
AND
DB_COS_ConfigMechanic(_Character, _Key, _Enabled)
AND
IntegerSubtract(1, _Enabled, _Next)
THEN
NOT DB_COS_ConfigMechanic(_Character, _Key, _Enabled);
DB_COS_ConfigMechanic(_Character, _Key, _Next);
PROC_COS_ConfigApplyMechanic(_Character, _Key, _Next);
PROC_COS_ConfigSyncMechanicMirrors(_Character);

PROC
PROC_COS_ConfigResetCore((CHARACTER)_Character)
AND
DB_COS_ConfigMechanicDefault(_Key, _Default)
AND
DB_COS_ConfigMechanic(_Character, _Key, _Enabled)
THEN
NOT DB_COS_ConfigMechanic(_Character, _Key, _Enabled);
DB_COS_ConfigMechanic(_Character, _Key, _Default);
PROC_COS_ConfigApplyMechanic(_Character, _Key, _Default);

IF
TutorialEvent(_Character, _Event)
AND
DB_COS_ConfigUiOpenedEvent(_Event)

PROC_COS_ConfigSyncCharacter(_Character);

IF

AND
DB_COS_ConfigMechanicEvent(_Event, _Key)
AND

AND
IsControlled(_Character, 1)
AND
IsInCombat(_Character, 0)
THEN
PROC_COS_ConfigToggleMechanic(_Character, _Key);

IF
TutorialEvent(_Character, _Event)
AND
DB_COS_ConfigResetCoreEvent(_Event)
AND
HasPassive(_Character, "COS_ChaosOriginMarker", 1)
AND
IsControlled(_Character, 1)
AND
IsInCombat(_Character, 0)
THEN
PROC_COS_ConfigResetCore(_Character);
PROC_COS_ConfigSyncMechanicMirrors(_Character);
