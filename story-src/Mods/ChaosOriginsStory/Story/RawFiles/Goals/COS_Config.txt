PROC_COS_ConfigApplyRacialPassive((CHARACTER)_Character, (STRING)_Passive)
AND

AND
HasPassive(_Character, _Passive, 0)
THEN
AddPassive(_Character, _Passive);
PROC_COS_ConfigRecordRacialGrant(_Character, _Passive);

PROC
PROC_COS_ConfigRecordRacialGrant((CHARACTER)_Character, (STRING)_Passive)
AND
HasPassive(_Character, _Passive, 1)
AND
NOT DB_COS_RacialPassiveGranted(_Character, _Passive)
THEN
DB_COS_RacialPassiveGranted(_Character, _Passive);

PROC
PROC_COS_ConfigApplyRacialPassive((CHARACTER)_Character, (STRING)_Passive)
AND
DB_COS_ConfigRacialPassive(_Character, _Passive, 0)
AND
DB_COS_RacialPassiveGranted(_Character, _Passive)
THEN
RemovePassive(_Character, _Passive);
NOT DB_COS_RacialPassiveGranted(_Character, _Passive);
