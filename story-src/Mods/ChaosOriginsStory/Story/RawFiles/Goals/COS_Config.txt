// Cost defaults/events are seeded at runtime so existing saves receive the new controls.
PROC
PROC_COS_ConfigSeedCosts()
THEN
DB_COS_ConfigCostDefault("Fate", 1, "COS_ConfigFateCost");
DB_COS_ConfigCostDefault("Genesis", 10, "COS_ConfigGenesisCost");
DB_COS_ConfigCostStepEvent((TUTORIALEVENT)COS_CFG_FATE_COST_MINUS_7d000000-0000-4000-8000-000000000001, "Fate", -1);
DB_COS_ConfigCostStepEvent((TUTORIALEVENT)COS_CFG_FATE_COST_PLUS_7d000000-0000-4000-8000-000000000002, "Fate", 1);
DB_COS_ConfigCostResetEvent((TUTORIALEVENT)COS_CFG_FATE_COST_RESET_7d000000-0000-4000-8000-000000000003, "Fate");
DB_COS_ConfigCostStepEvent((TUTORIALEVENT)COS_CFG_GENESIS_COST_MINUS_7d000000-0000-4000-8000-000000000004, "Genesis", -1);
DB_COS_ConfigCostStepEvent((TUTORIALEVENT)COS_CFG_GENESIS_COST_PLUS_7d000000-0000-4000-8000-000000000005, "Genesis", 1);
DB_COS_ConfigCostResetEvent((TUTORIALEVENT)COS_CFG_GENESIS_COST_RESET_7d000000-0000-4000-8000-000000000006, "Genesis");

PROC
PROC_COS_ConfigEnsureCosts((CHARACTER)_Character)
THEN
PROC_COS_ConfigSeedCosts();
PROC_COS_ConfigInitCosts(_Character);

THEN

PROC

AND
DB_COS_ConfigCostResetEvent(_Event, _Key)
THEN

PROC
PROC_COS_ConfigSyncCosts((CHARACTER)_Character)
THEN
AddPassive(_Character, "COS_CFG_COST_CARRIER");
PROC_COS_ConfigSyncCostValues(_Character);

PROC
PROC_COS_ConfigSyncCostValues((CHARACTER)_Character)
AND
DB_COS_ConfigCost(_Character, _Key, _Value)
AND
DB_COS_ConfigCostDefault(_Key, _Default, _Resource)
AND
IntegerToReal(_Value, _Real)
AND
RealSubtract(0.0, 1000.0, _Reset)
THEN
PartyIncreaseActionResourceValue(_Character, _Resource, _Reset);
PartyIncreaseActionResourceValue(_Character, _Resource, _Real);

PROC
PROC_COS_ConfigSetCost((CHARACTER)_Character, (STRING)_Key, (INTEGER)_Value)
AND
_Value >= 0
AND
_Value <= 20
AND
DB_COS_ConfigCostDefault(_Key, _Default, _Resource)
AND
DB_COS_ConfigCost(_Character, _Key, _Old)
THEN
NOT DB_COS_ConfigCost(_Character, _Key, _Old);
DB_COS_ConfigCost(_Character, _Key, _Value);
PROC_COS_ConfigSyncCosts(_Character);
PROC_COS_SyncPowerFromDatabase(_Character);

PROC
PROC_COS_ConfigStepCost((CHARACTER)_Character, (STRING)_Key, (INTEGER)_Delta)
AND
DB_COS_ConfigCost(_Character, _Key, _Old)
AND
IntegerSum(_Old, _Delta, _RawValue)
AND
IntegerMax(_RawValue, 0, _FloorValue)
AND
IntegerMin(_FloorValue, 20, _Value)
THEN
PROC_COS_ConfigSetCost(_Character, _Key, _Value);

DB_COS_ConfigCostResetEvent(_Event, _Key)

AND

AND
