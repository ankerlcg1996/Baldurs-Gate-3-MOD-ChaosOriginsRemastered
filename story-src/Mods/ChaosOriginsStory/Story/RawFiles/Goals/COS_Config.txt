PROC
PROC_COS_ConfigApplyMechanic((CHARACTER)_Character, (STRING)_Key)
AND
DB_COS_ConfigMechanic(_Character, _Key, 1)
AND

AND

AND
_Level >= _RequiredLevel
AND
