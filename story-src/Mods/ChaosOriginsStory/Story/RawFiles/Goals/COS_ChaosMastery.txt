PROC
PROC_COS_EnsureMasteryCounts((CHARACTER)_Character)
AND
NOT DB_COS_MasteryAvailableCount(_Character, _)
THEN
DB_COS_MasteryAvailableCount(_Character, 0);

PROC
PROC_COS_EnsureMasteryCounts((CHARACTER)_Character)
AND
NOT DB_COS_MasteryTuneCount(_Character, _)
THEN
DB_COS_MasteryTuneCount(_Character, 0);

PROC
PROC_COS_EnsureMasteryCounts((CHARACTER)_Character)
AND
NOT DB_COS_MasteryCorrectCount(_Character, _)
THEN
DB_COS_MasteryCorrectCount(_Character, 0);

PROC
PROC_COS_ApplyMasteryRouteStatus((CHARACTER)_Character, (STRING)_Status, (INTEGER)_Count)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
_Count > 0
AND
IntegerProduct(_Count, 6, _DurationSeconds)
AND
IntegerToReal(_DurationSeconds, _Duration)
THEN
ApplyStatus(_Character, _Status, _Duration, 100, _Character);

PROC
PROC_COS_SyncMasteryStatuses((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
DB_COS_MasteryTuneCount(_Character, _TuneCount)
AND
DB_COS_MasteryCorrectCount(_Character, _CorrectCount)
THEN
RemoveStatus(_Character, "COS_CHAOS_MASTERY_TUNE", _Character);
RemoveStatus(_Character, "COS_CHAOS_MASTERY_CORRECT", _Character);
PROC_COS_ApplyMasteryRouteStatus(_Character, "COS_CHAOS_MASTERY_TUNE", _TuneCount);
PROC_COS_ApplyMasteryRouteStatus(_Character, "COS_CHAOS_MASTERY_CORRECT", _CorrectCount);
PROC_COS_SyncOverview(_Character);

PROC
PROC_COS_AccrueMastery((CHARACTER)_Character)
AND
HasPassive(_Character, "COS_ChaosOriginMarker", 1)
AND
HasPassive(_Character, "COS_ChaosMasteryGuide", 0)
THEN
AddPassive(_Character, "COS_ChaosMasteryGuide");

PROC
PROC_COS_AccrueMastery((CHARACTER)_Character)
AND
HasPassive(_Character, "COS_ChaosOriginMarker", 1)
AND
DB_COS_MasterySchema46To47(_Character)
THEN
PROC_COS_SyncMasteryAfterSchema47(_Character);

PROC
PROC_COS_AccrueMastery((CHARACTER)_Character)
AND
HasPassive(_Character, "COS_ChaosOriginMarker", 1)
AND
NOT DB_COS_MasterySchema46To47(_Character)
THEN
PROC_COS_MigrateMasterySchema46To47(_Character);

PROC
PROC_COS_SyncMastery((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
DB_COS_MasteryTuneCount(_Character, _)
AND
DB_COS_MasteryCorrectCount(_Character, _)
THEN
PROC_COS_SyncMasteryStatuses(_Character);
PROC_COS_UpdateMasterySpell(_Character);

PROC
PROC_COS_MigrateMasterySchema46To47((CHARACTER)_Character)
AND
NOT DB_COS_MasterySchema46To47(_Character)
THEN
PROC_COS_ResetMastery(_Character);

PROC
PROC_COS_SyncMasteryAfterSchema47((CHARACTER)_Character)
AND
DB_COS_MasterySchema46To47(_Character)
AND
GetLevel(_Character, _Level)
AND
IntegerMin(_Level, 12, _Cap)
THEN
PROC_COS_EnsureMasteryCounts(_Character);
PROC_COS_GrantMasteryFrom(_Character, 1, _Cap);

PROC
PROC_COS_IncrementMasteryAvailable((CHARACTER)_Character)
AND
DB_COS_MasteryAvailableCount(_Character, _Count)
AND
IntegerSum(_Count, 1, _Next)
THEN
NOT DB_COS_MasteryAvailableCount(_Character, _Count);
DB_COS_MasteryAvailableCount(_Character, _Next);

PROC
PROC_COS_ConsumeMasteryAvailable((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
DB_COS_MasteryAvailableCount(_Character, _Count)
AND
_Count > 0
AND
IntegerSubtract(_Count, 1, _Next)
THEN
NOT DB_COS_MasteryAvailableCount(_Character, _Count);
DB_COS_MasteryAvailableCount(_Character, _Next);
PROC_COS_UpdateMasterySpell(_Character);

PROC
PROC_COS_GrantMasteryFrom((CHARACTER)_Character, (INTEGER)_Cursor, (INTEGER)_Cap)
AND
_Cursor <= _Cap
AND
DB_COS_MasteryCarrier(_Cursor, _Carrier)
AND
NOT DB_COS_MasteryEarned(_Character, _Cursor)
AND
HasPassive(_Character, _Carrier, 0)
AND
IntegerSum(_Cursor, 1, _Next)
THEN
AddPassive(_Character, _Carrier);
DB_COS_MasteryEarned(_Character, _Cursor);
PROC_COS_IncrementMasteryAvailable(_Character);
PROC_COS_GrantMasteryFrom(_Character, _Next, _Cap);

PROC
PROC_COS_GrantMasteryFrom((CHARACTER)_Character, (INTEGER)_Cursor, (INTEGER)_Cap)
AND
_Cursor <= _Cap
AND
DB_COS_MasteryEarned(_Character, _Cursor)
AND
IntegerSum(_Cursor, 1, _Next)
THEN
PROC_COS_GrantMasteryFrom(_Character, _Next, _Cap);

PROC
PROC_COS_GrantMasteryFrom((CHARACTER)_Character, (INTEGER)_Cursor, (INTEGER)_Cap)
AND
_Cursor > _Cap
THEN
PROC_COS_UpdateMasterySpell(_Character);

PROC
PROC_COS_UpdateMasterySpell((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
DB_COS_MasteryAvailableCount(_Character, _Count)
AND
_Count > 0
AND
HasSpell(_Character, "Shout_COS_ChaosMastery", 0)
THEN
AddSpell(_Character, "Shout_COS_ChaosMastery", 0, 1);

PROC
PROC_COS_UpdateMasterySpell((CHARACTER)_Character)
AND
DB_COS_ConfigMechanic((CHARACTER)_Character, "Mastery", 1)
AND
DB_COS_MasteryAvailableCount(_Character, 0)
AND
HasSpell(_Character, "Shout_COS_ChaosMastery", 1)
THEN
RemoveSpell(_Character, "Shout_COS_ChaosMastery", 1);

PROC
PROC_COS_ClearMasteryAvailableCount((CHARACTER)_Character)
AND
DB_COS_MasteryAvailableCount(_Character, _Count)
THEN
NOT DB_COS_MasteryAvailableCount(_Character, _Count);

PROC
PROC_COS_ClearMasteryTuneCount((CHARACTER)_Character)
AND
DB_COS_MasteryTuneCount(_Character, _Count)
THEN
NOT DB_COS_MasteryTuneCount(_Character, _Count);

PROC
PROC_COS_ClearMasteryCorrectCount((CHARACTER)_Character)
AND
DB_COS_MasteryCorrectCount(_Character, _Count)
THEN
NOT DB_COS_MasteryCorrectCount(_Character, _Count);

PROC
PROC_COS_ResetMastery((CHARACTER)_Character)
THEN
DB_COS_MasterySchema46To47(_Character);
RemoveStatus(_Character, "COS_CHAOS_MASTERY_TUNE", _Character);
RemoveStatus(_Character, "COS_CHAOS_MASTERY_CORRECT", _Character);
PROC_COS_ResetMasteryCarrier(_Character, 1);

PROC
PROC_COS_ResetMasteryCarrier((CHARACTER)_Character, (INTEGER)_Cursor)
AND
_Cursor <= 12
AND
DB_COS_MasteryCarrier(_Cursor, _Carrier)
AND
HasPassive(_Character, _Carrier, 1)
AND
IntegerSum(_Cursor, 1, _Next)
THEN
RemovePassive(_Character, _Carrier);
NOT DB_COS_MasteryEarned(_Character, _Cursor);
PROC_COS_ResetMasteryCarrier(_Character, _Next);

PROC
PROC_COS_ResetMasteryCarrier((CHARACTER)_Character, (INTEGER)_Cursor)
AND
_Cursor <= 12
AND
DB_COS_MasteryCarrier(_Cursor, _Carrier)
AND
HasPassive(_Character, _Carrier, 0)
AND
IntegerSum(_Cursor, 1, _Next)
THEN
NOT DB_COS_MasteryEarned(_Character, _Cursor);
PROC_COS_ResetMasteryCarrier(_Character, _Next);

PROC
PROC_COS_ResetMasteryCarrier((CHARACTER)_Character, (INTEGER)_Cursor)
AND
_Cursor > 12
THEN
PROC_COS_ClearMasteryAvailableCount(_Character);
PROC_COS_ClearMasteryTuneCount(_Character);
PROC_COS_ClearMasteryCorrectCount(_Character);
DB_COS_MasteryAvailableCount(_Character, 0);
DB_COS_MasteryTuneCount(_Character, 0);
DB_COS_MasteryCorrectCount(_Character, 0);
PROC_COS_SyncMasteryAfterSchema47(_Character);
