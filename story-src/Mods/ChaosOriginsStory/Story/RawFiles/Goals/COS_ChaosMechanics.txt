PROC_COS_ApplyPowerDisplay((CHARACTER)_Character, (INTEGER)_Power)
AND
_Power > 0
AND
IntegerProduct(_Power, 6, _DurationSeconds)
AND
IntegerToReal(_DurationSeconds, _Duration)
THEN
ApplyStatus(_Character, "COS_CHAOS_POWER_STACK", _Duration, 100, _Character);

PROC_COS_ApplyPowerDisplay((CHARACTER)_Character, (INTEGER)_Power)
AND
_Power == 0
THEN
ApplyStatus(_Character, "COS_CHAOS_POWER_STACK", -1.0, 100, _Character);

AND
_Power >= 10
THEN
ApplyStatus(_Character, "COS_CHAOS_GENESIS_READY", -1.0, 100, _Character);

_OldCount < 10
AND
IntegerSum(_OldCount, 1, _NewCount)

AND
Random(100, _Roll)
THEN
NOT DB_COS_LostCount(_Character, _OldCount);
DB_COS_LostCount(_Character, _NewCount);
PROC_COS_SyncLostDisplay(_Character, _NewCount);
PROC_COS_ConvertLost(_Character, _NewCount, _Chance, _Roll);
